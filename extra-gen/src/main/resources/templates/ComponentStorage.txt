package #pkg

import com.cws.extra.ecs.*
import com.cws.extra.memory.*

val #T.Companion.COMPONENT_ID: Int get() = #COMPONENT_ID

val Scene.#extension: #TList get() = registry[#T.COMPONENT_ID].list as #TList

inline fun #TStorage(capacity: Int) = #TStorage(
    #TList(capacity),
    ComponentPool(capacity),
)

@ExtraData
data class #TStorage(
    override val list: #TList,
    val pool: ComponentPool,
) : ComponentStorage {

    override fun <T> add(entity: Int, component: T) {
        val c = component as? #T ?: return
        val componentIndex = pool.add(entity)
        if (componentIndex == list.size) {
            list.add(c)
        } else {
            list[componentIndex] = c
        }
    }

    override fun remove(entity: Int) {
        if (pool.has(entity)) {
            val index = pool.remove(entity)
            if (index != ComponentNull) {
                list.removeAtSwap(index)
            }
        }
    }

    override fun clear() {
        list.clear()
        pool.clear()
    }

    override fun has(entity: Int) = pool.has(entity)

    override fun index(entity: Int) = pool.getComponentIndex(entity)

}
